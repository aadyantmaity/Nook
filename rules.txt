rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // USERS
    match /users/{userId} {
      allow read: if request.auth != null;
      allow create: if request.auth.uid == userId;
      allow delete: if request.auth != null && request.auth.uid == userId;

      // Allow users to update their own profile (all fields)
      allow update: if request.auth.uid == userId;

      // Allow follow/unfollow operations
      allow update: if request.auth != null
        && request.writeFields.hasOnly(['followers']);

      // Users can add/remove others from their following list
      allow update: if request.auth.uid == userId
        && request.writeFields.hasOnly(['following']);

      // Allow updating checkedIn field (check-in tracking)
      allow update: if request.auth.uid == userId
        && request.writeFields.hasOnly(['checkedIn']);

      // RECENT ACTIVITY SUBCOLLECTION
      match /recentActivity/{activityId} {
        allow read: if request.auth != null && request.auth.uid == userId;

        allow create: if request.auth != null && (
          // Follow event: someone followed this user
          (request.resource.data.type == 'follow' &&
           request.resource.data.followedBy == request.auth.uid &&
           request.resource.data.keys().hasAll(['timestamp', 'followedBy', 'type'])) ||

          // Check-in event: someone checked into this user's nook
          (request.resource.data.type == 'checkin' &&
           request.resource.data.checkedInBy == request.auth.uid &&
           request.resource.data.keys().hasAll(['timestamp', 'checkedInBy', 'type', 'spotId', 'spotName']) &&
           request.resource.data.timestamp is timestamp &&
           request.resource.data.checkedInBy is string &&
           request.resource.data.type is string &&
           request.resource.data.spotId is string &&
           request.resource.data.spotName is string)
        );

        // Allow updating recent activity (for adding displayName to check-in events)
        allow update: if request.auth != null && (
          request.writeFields.hasOnly(['displayName']) &&
          resource.data.type == 'checkin' &&
          resource.data.checkedInBy == request.auth.uid
        );

        // Allow deleting old activity entries
        allow delete: if request.auth != null && request.auth.uid == userId;
      }

      // FOCUS SESSIONS
      match /focusSessions/{sessionId} {
        allow create: if request.auth != null && request.auth.uid == userId;
        allow read: if request.auth != null;
        allow delete: if request.auth != null && request.auth.uid == userId;
        allow update: if false;
      }

      // FOCUS GOALS
      match /focusGoals/{goalId} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
        allow create: if request.auth != null
          && request.auth.uid == userId
          && request.resource.data.keys().hasAll(['text', 'createdAt', 'completed', 'archived', 'progress', 'target'])
          && request.resource.data.text is string
          && request.resource.data.text.size() > 0
          && request.resource.data.text.size() <= 200
          && request.resource.data.completed is bool
          && request.resource.data.archived is bool
          && request.resource.data.progress is number
          && request.resource.data.target is number
          && request.resource.data.progress >= 0
          && request.resource.data.target > 0
          && request.resource.data.progress <= request.resource.data.target;
        allow update: if request.auth != null
          && request.auth.uid == userId
          && (
            request.writeFields.hasOnly(['progress', 'completed', 'completedAt']) ||
            request.writeFields.hasOnly(['archived']) ||
            request.writeFields.hasOnly(['text']) ||
            request.writeFields.hasOnly(['target'])
          );
      }

      // FOCUS TASKS
      match /focusTasks/{taskId} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
        allow create: if request.auth != null
          && request.auth.uid == userId
          && request.resource.data.keys().hasAll(['text', 'createdAt', 'completed', 'archived'])
          && request.resource.data.text is string
          && request.resource.data.text.size() > 0
          && request.resource.data.text.size() <= 200
          && request.resource.data.completed is bool
          && request.resource.data.archived is bool;
        allow update: if request.auth != null
          && request.auth.uid == userId;
      }
    }

    // STUDY SPOTS
    match /studySpots/{spotId} {
      allow read: if true;
      allow create: if request.auth != null &&
                    request.resource.data.addedBy == request.auth.uid;
      allow update: if request.auth != null && (
        request.auth.uid == resource.data.addedBy
        ||
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['locationAlerts'])
        ||
        (
          resource.data.sharedWith != null &&
          resource.data.sharedWith.hasAny([request.auth.uid]) &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['privateNotes'])
        )
      );
      allow delete: if request.auth != null &&
                    request.auth.uid == resource.data.addedBy;

      // NOTES SUBCOLLECTION
      match /notes/{noteId} {
        allow read, create, delete: if request.auth != null &&
          request.auth.uid == get(/databases/$(database)/documents/studySpots/$(spotId)).data.addedBy;
      }

      // RATINGS SUBCOLLECTION
      match /ratings/{ratingId} {
        allow read, create: if request.auth != null;
      }

      // FAVORITED BY SUBCOLLECTION
      match /favoritedBy/{userId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null &&
                      request.auth.uid == userId &&
                      request.resource.data.keys().hasAll(['favoritedAt', 'userId']) &&
                      request.resource.data.userId == userId;
        allow delete: if request.auth != null && request.auth.uid == userId;
      }
    }

    // OVERALL RATINGS (top-level collection)
    match /overallRatings/{spotId} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // SPOT FEEDBACK
    match /spotFeedback/{feedbackId} {
      allow read: if false;
      allow create: if request.auth != null;
      allow update, delete: if false;
    }

    // REPORTS
    match /reports/{reportId} {
      allow create: if request.auth != null;
      allow read, update, delete: if false;
    }

    // GOAL ARCHIVES
    match /goalArchives/{userId} {
      allow read: if request.auth != null && request.auth.uid == userId;
      allow create: if request.auth != null
        && request.auth.uid == userId
        && request.resource.data.keys().hasAll(['goals'])
        && request.resource.data.goals is list;
      allow update: if request.auth != null
        && request.auth.uid == userId
        && request.writeFields.hasOnly(['goals']);
      allow delete: if request.auth != null && request.auth.uid == userId;
    }

    // GALLERY IMAGES
    match /galleryImages/{spotId}/images/{imageId} {
      allow read: if true;
      allow create: if request.auth != null && request.resource.data.authorId == request.auth.uid;
      allow delete: if request.auth != null && resource.data.authorId == request.auth.uid;
    }

    // STUDY GROUPS
    match /studyGroups/{groupId} {
      // Allow read if user is a member or group is public
      allow read: if request.auth != null && (
        resource.data.isPrivate == false ||
        resource.data.memberUids.hasAny([request.auth.uid])
      );

      // Allow create if user is included in the memberUids
      allow create: if request.auth != null &&
        request.resource.data.memberUids.hasAny([request.auth.uid]);

      // Allow update with various permissions
      allow update: if request.auth != null && (
        // 1. Owner can update any fields
        resource.data.createdBy == request.auth.uid ||

        // 2. Admin can update name, description, memberUids, adminUids
        (resource.data.adminUids.hasAny([request.auth.uid]) &&
          (
            request.writeFields.hasOnly(['name']) ||
            request.writeFields.hasOnly(['description']) ||
            request.writeFields.hasOnly(['name', 'description']) ||
            request.writeFields.hasOnly(['memberUids']) ||
            (request.writeFields.hasOnly(['adminUids']) &&
              request.resource.data.adminUids.hasAny([resource.data.createdBy]))
          )
        ) ||

        // 3. Member removing themselves
        (resource.data.memberUids.hasAny([request.auth.uid]) &&
          request.resource.data.memberUids.size() == resource.data.memberUids.size() - 1 &&
          !request.resource.data.memberUids.hasAny([request.auth.uid])
        ) ||

        // 4. Member adding new members
        (resource.data.memberUids.hasAny([request.auth.uid]) &&
          request.resource.data.memberUids.size() >= resource.data.memberUids.size() &&
          resource.data.memberUids.hasAll(request.resource.data.memberUids)
        ) ||

        // 5. Public group, user adding themselves
        (resource.data.isPrivate == false &&
          !resource.data.memberUids.hasAny([request.auth.uid]) &&
          request.resource.data.memberUids.size() == resource.data.memberUids.size() + 1 &&
          request.resource.data.memberUids.hasAll(resource.data.memberUids) &&
          request.resource.data.memberUids.hasAny([request.auth.uid])
        )
      );

      // Allow delete if user is the creator
      allow delete: if request.auth != null &&
        resource.data.createdBy == request.auth.uid;

      // Messages subcollection
      match /messages/{messageId} {
        allow read: if request.auth != null &&
          (
            get(/databases/$(database)/documents/studyGroups/$(groupId)).data.isPrivate == false ||
            get(/databases/$(database)/documents/studyGroups/$(groupId)).data.memberUids.hasAny([request.auth.uid])
          );
        allow create: if request.auth != null &&
          get(/databases/$(database)/documents/studyGroups/$(groupId)).data.memberUids.hasAny([request.auth.uid]);
        allow delete: if request.auth != null &&
          (resource.data.uid == request.auth.uid ||
           get(/databases/$(database)/documents/studyGroups/$(groupId)).data.createdBy == request.auth.uid);
        allow update: if request.auth != null &&
          get(/databases/$(database)/documents/studyGroups/$(groupId)).data.memberUids.hasAny([request.auth.uid]) &&
          request.writeFields.hasOnly(["poll.votes." + request.auth.uid]);
      }

      // Tasks subcollection
      match /tasks/{taskId} {
        allow read: if request.auth != null &&
          (
            get(/databases/$(database)/documents/studyGroups/$(groupId)).data.isPrivate == false ||
            get(/databases/$(database)/documents/studyGroups/$(groupId)).data.memberUids.hasAny([request.auth.uid])
          );
        allow create: if request.auth != null &&
          get(/databases/$(database)/documents/studyGroups/$(groupId)).data.memberUids.hasAny([request.auth.uid]);
        allow update: if request.auth != null &&
          get(/databases/$(database)/documents/studyGroups/$(groupId)).data.memberUids.hasAny([request.auth.uid]) &&
          (
            request.writeFields.hasOnly(['checkedOff']) &&
            (
              (!resource.data.checkedOff.hasAny([request.auth.uid]) && request.resource.data.checkedOff.hasAny([request.auth.uid]))
              ||
              (resource.data.checkedOff.hasAny([request.auth.uid]) && !request.resource.data.checkedOff.hasAny([request.auth.uid]))
            )
          );
        allow delete: if get(/databases/$(database)/documents/studyGroups/$(groupId)).data.createdBy == request.auth.uid;
      }
    }
  }
}
